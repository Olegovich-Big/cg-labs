cmake_minimum_required(VERSION 3.20)
project(CG_Labs LANGUAGES CXX)
if(NOT WIN32)
  message(FATAL_ERROR "These laboratory applications use Windows and OpenGL.")
endif()
set(CMAKE_CXX_STANDARD 17)
foreach(n RANGE 1 11)
  if(n LESS 10)
    set(id "0${n}")
  else()
    set(id "${n}")
  endif()
  add_executable(lab${id} src/lab${id}.cpp)
  target_compile_definitions(lab${id} PRIVATE UNICODE _UNICODE NOMINMAX)
  target_link_libraries(lab${id} PRIVATE opengl32 glu32 gdiplus comdlg32 user32 gdi32 shell32)
  if(MSVC)
    target_compile_options(lab${id} PRIVATE /W4 /utf-8 /EHsc)
  endif()
endforeach()
